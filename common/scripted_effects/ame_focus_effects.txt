AME_initialize_republic = {
	set_country_flag = AME_restored_republic
	if = {
		limit = { NOT = { has_country_leader = { ruling_only = yes } } }
		set_popularities = {
			democratic = 90
			fascism = 5
			communism = 5
			neutrality = 0
		}
		set_politics = {
			ruling_party = democratic
			elections_allowed = yes
		}
		create_country_leader = {
			name = "AME_provisional_government"
			desc = "AME_provisional_government_desc"
			picture = "gfx/leaders/Europe/Portrait_Europe_Generic_land_1.dds"
			expire = "2051.1.1"
			ideology = liberalism
			traits = { }
		}
	}
}
