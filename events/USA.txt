# Mexico receives the American security offer.
country_event = {
	id = usa2.1
	title = usa2.1.t
	desc = usa2.1.d
	picture = GFX_report_event_hol_cabinet_de_geer

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = usa2.1.a
		ai_chance = {
			base = 60
		}

		USA = {
			if = {
				limit = {
					is_faction_leader = yes
					MEX = { is_in_faction = no }
				}
				add_to_faction = MEX
			}
			else_if = {
				limit = {
					is_in_faction_with = AME
					MEX = { is_in_faction = no }
				}
				AME = { add_to_faction = MEX }
			}
			set_country_flag = USA_mexico_allied
		}

		hidden_effect = {
			news_event = { hours = 6 id = news.403 }
		}
	}

	option = {
		name = usa2.1.b
		ai_chance = {
			base = 25
		}

		USA = {
			add_political_power = 25
		}
	}

	option = {
		name = usa2.1.c
		ai_chance = {
			base = 15
		}

		USA = {
			set_country_flag = USA_mexico_rejected
		}

		hidden_effect = {
			news_event = { hours = 6 id = news.404 }
		}
	}
}

# The constitutional crisis.
country_event = {
	id = usa2.7
	title = usa2.7.t
	desc = usa2.7.d
	picture = GFX_report_event_hol_cabinet_de_geer

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = usa2.7.a
		ai_chance = {
			base = 50
		}

		if = {
			limit = { has_idea = USA_presidential_emergency_powers }
			remove_ideas = USA_presidential_emergency_powers
		}
		if = {
			limit = { has_idea = public_objection }
			remove_ideas = public_objection
		}
		set_country_flag = USA_constitutional_crisis_resolved
		add_stability = 0.05
		add_political_power = -50
	}

	option = {
		name = usa2.7.b
		ai_chance = {
			base = 50
		}

		set_country_flag = USA_constitutional_crisis_escalated
		add_war_support = 0.05
		add_political_power = 50
		add_popularity = {
			ideology = fascism
			popularity = 0.15
		}
		add_popularity = {
			ideology = democratic
			popularity = -0.15
		}
	}
}

# AME decides whether to admit the United States.
country_event = {
	id = usa2.10
	title = usa2.10.t
	desc = usa2.10.d
	picture = GFX_report_event_hol_cabinet_de_geer

	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = usa2.10.a
		ai_chance = {
			base = 70
		}

		if = {
			limit = { USA = { is_in_faction = no } }
			add_to_faction = USA
		}
		if = {
			limit = {
				USA = { has_country_flag = USA_mexico_allied }
				country_exists = MEX
				MEX = { is_in_faction = no }
			}
			add_to_faction = MEX
		}
		USA = {
			set_country_flag = USA_allied_member
		}
	}

	option = {
		name = usa2.10.b
		ai_chance = {
			base = 30
		}

		USA = {
			set_country_flag = USA_allied_membership_refused
			if = {
				limit = { is_in_faction = no }
				create_faction = american_coalition
			}
			if = {
				limit = {
					has_country_flag = USA_mexico_allied
					country_exists = MEX
					MEX = { is_in_faction = no }
				}
				add_to_faction = MEX
			}
		}
	}
}
