country_event = {
	id = ame.3
	title = ame.3.t
	desc = ame.3.d
	picture = GFX_news_event_ast_planning
	
	is_triggered_only = yes
	
	option = { # Accept
		name = ame.3.a
		ai_chance = {
			base = 30
			modifier = {
				add = -25
				AME = { has_army_size = { size < 30 } }
			}
			modifier = {
				add = 100
				GER = { has_army_size = { size > 29 } }
			}
			modifier = {
				add = 20
				is_in_faction_with = AME
			}
		}
		AME = {
			country_event = { hours = 2 id = ame.1 }
		}
		custom_effect_tooltip = GAME_OVER_TT
	}
	option = { # Refuse
		name = ame.3.b
		ai_chance = {
			base = 10			
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
			modifier = {
				AME = { has_army_size = { size > 29 } }
				factor = 0.5
			}
			modifier = {
				add = 20
				AME = { has_army_size = { size < 10 } }
			}
			#modifier = {
			#	add = 100
			#	HUN = { 
			#		OR = { 
			#			has_completed_focus = HUN_protect_austria 
			#			is_in_faction_with = AUS
			#		}
			#	}
			#}
			modifier = {
				factor = 2
				is_in_faction = yes
				NOT = { is_in_faction_with = AME }
			}
			#modifier = {
			#	any_other_country = {
			#		has_guaranteed = ENG
			#	}
			#	factor = 5
			#}
		}
		AME = {
			country_event = { hours = 2 id = ame.2 }
			if = { limit = { is_in_faction_with = ENG }
				remove_from_faction = ENG
			}
		}
	}
}
